document.addEventListener("DOMContentLoaded", async () => {
  const selectBox = document.getElementById("selectBox");
  const selectedOption = document.getElementById("selectedOption");
  const dropdown = document.querySelector(".cities-select__dropdown");
  const searchInput = document.getElementById("searchInput");
  const optionsList = document.getElementById("optionsList");
	const dropdownIcon = document.querySelector(".cities-select__dropdown-icon");
  
  document.querySelector("#citiesSelectForm").addEventListener("keydown", function (event) {
    if (event.key === "Enter" && event.target.tagName !== "TEXTAREA") {
      event.preventDefault();
    }
  });

	const flagBigMarketModal = () => {
    const elements = document.querySelectorAll('[data-modal-city-segment="big-market"]');

    for (const el of elements) {
      const style = window.getComputedStyle(el);
      if (style.display === 'block' || style.display === 'flex') {
        el.setAttribute('data-modal', 'active');
        break;
      }
    }
	}

  let allCities = [];

  const defaultCities = [
    { nome: "São Paulo", uf: "SP" },
    { nome: "Rio de Janeiro", uf: "RJ" },
    { nome: "Belo Horizonte", uf: "MG" },
    { nome: "Brasília", uf: "DF" },
    { nome: "Curitiba", uf: "PR" },
  ];

	const formatCityName = (city) => {
		return city
      .normalize("NFD")
      .replace(/[\u0300-\u036f]/g, "")
      .toLowerCase()
      .replace(/ /g, "-");
	}

	const fetchWebflowCollectionItem = async (slug) => {
    const url = 'https://api.escale.com.br/escale-os-bff';
		// const url = 'https://api.staging.escale.com.br/escale-os-bff';
		const id = '67b4ec7bc395883c862f6520' 

		const formattedSlug = formatCityName(slug)

    const options = {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
      },
      body: JSON.stringify({
        query: `
          query WebflowCollectionItemBySlug($id: String!, $slug: String!) {
            webflowCollectionItemBySlug(id: $id, slug: $slug) {
              error
              data {
                id
                fieldData {
                  name
                  slug
                  regional
                  tipoDeCidade
                  grupoOferta
                }
              }
            }
          }
        `,
        operationName: 'WebflowCollectionItemBySlug',
        variables: { id, slug: formattedSlug },
      }),
    };

    try {
      const response = await fetch(url, options);
      const data = await response.json();
			const queryData = data.data.webflowCollectionItemBySlug
			if(queryData) {
				return queryData.data.fieldData.slug === formattedSlug
			} else {
				return false
			}
    } catch (error) {
      return false;
    }
  }

  const togglePricingDisplay = async (city, loader = true) => {
		const screenLoader = document.querySelector(".cities-select__loader-overlay");
		flagBigMarketModal();

		if(loader) {
			screenLoader.style.display = "flex";
		}
		
		const showAlternativePricing = await fetchWebflowCollectionItem(city);

    document
      .querySelectorAll("[data-city-segment='development-market']")
      .forEach((el) => (el.style.display = showAlternativePricing ? "block" : "none"));
    document
      .querySelectorAll("[data-city-segment='big-market']")
      .forEach((el) => (el.style.display = showAlternativePricing ? "none" : "block"));
    document
      .querySelectorAll("[data-modal-city-segment='big-market'][data-modal='active']")
      .forEach((el) => (el.style.display = showAlternativePricing ? "none" : "block"));
    document
      .querySelectorAll("#banner-comun-market")
      .forEach((el) => (el.style.display = showAlternativePricing ? "none" : "block"));
    document
      .querySelectorAll("#banner-development-market")
      .forEach((el) => (el.style.display = showAlternativePricing ? "block" : "none"));
    document
      .querySelectorAll(".banner__price")
      .forEach((el) => (el.style.display = showAlternativePricing ? "none" : "inline-flex"));
    document
      .querySelectorAll(".banner__price--development-market")
      .forEach((el) => (el.style.display = showAlternativePricing ? "inline-flex" : "none"));
    document
      .querySelectorAll(".banner-button__description-price.banner__price")
      .forEach((el) => (el.style.display = showAlternativePricing ? "none" : "inline"));
    document
      .querySelectorAll(".banner-button__description-price.banner__price--development-market")
      .forEach((el) => (el.style.display = showAlternativePricing ? "inline" : "none"));

		if(loader) {
			screenLoader.style.display = "none";
		}
  };

  const populateOptions = (cities) => {
    optionsList.innerHTML = "";

    if (cities.length === 0) {
      const noOption = document.createElement("li");
      noOption.classList.add("cities-select__no-result");
      noOption.textContent = "Nenhuma cidade encontrada";
      optionsList.appendChild(noOption);
      return;
    }

    cities.forEach(({ nome, uf }) => {
      const option = document.createElement("li");
			const cityName = formatCityName(`${nome}-${uf}`)
      option.classList.add(`cities-select__option`, `track-cities-select-option-${cityName}`);
      option.setAttribute("tabindex", "-1");
      option.setAttribute("role", "option");
      option.textContent = `${nome} - ${uf}`;
      option.addEventListener("click", () => {
        selectedOption.textContent = option.textContent;
        dropdown.style.display = "none";
        dropdownIcon.style.transform = "rotate(180deg)";
        togglePricingDisplay(nome);
      });

      optionsList.appendChild(option);
    });
  };

  const getCityFromCoords = (latitude, longitude) => {
    const url = `https://nominatim.openstreetmap.org/reverse?lat=${latitude}&lon=${longitude}&format=json`;
  
    fetch(url)
      .then((response) => response.json())
      .then((data) => {
        const city =
          data.address.city || data.address.town || data.address.village;

        togglePricingDisplay(city)
        const selectedCity = allCities.find(currentCity => {
          const currentFormattedCity = formatCityName(currentCity?.nome || '')
          
          return currentFormattedCity === formatCityName(city)
        })

        selectedOption.textContent = selectedCity ? `${city} - ${selectedCity.uf}` : city
      })
      .catch((error) => console.error("Error getting location data:", error));
  }

  const getLocation = () => {
    if (!navigator.geolocation) return;
  
    navigator.geolocation.getCurrentPosition(
      (position) => {
      	getCityFromCoords(position.coords.latitude, position.coords.longitude);
      }
    );
  }

  const checkGeolocationPermission = () => {
    navigator.permissions.query({ name: "geolocation" }).then((result) => {
      console.log("Current permission state:", result.state);
  
      populateOptions(defaultCities);
      
      if (result.state === "granted" || result.state === "prompt") getLocation();
  
      result.onchange = () => {
        if (result.state !== "granted" && result.state !== "prompt") return;
        getLocation();
      };
    });
  }

  const fetchAllCities = async () => {
    try {
      const response = await fetch("https://servicodados.ibge.gov.br/api/v1/localidades/municipios");
      const data = await response.json();
      allCities = data.map(({ nome, microrregiao }) => ({
        nome,
        uf: microrregiao && microrregiao.mesorregiao ? microrregiao.mesorregiao?.UF?.sigla : '',
      }));
    } catch (error) {
      console.error("Erro ao buscar cidades:", error);
    }
  };

  const searchCities = (searchText) => {
    if (!searchText) {
      populateOptions(defaultCities);
      return;
    }

    const filteredCities = allCities
      .filter(({ nome }) => formatCityName(nome).includes(formatCityName(searchText)))
      .slice(0, 10);

    populateOptions(filteredCities);
  };

  const debounce = (func, delay = 300) => {
    let timeout;
    return (...args) => {
      clearTimeout(timeout);
      timeout = setTimeout(() => func(...args), delay);
    };
  };

  selectBox.addEventListener("click", () => {
  	dropdownIcon.style.transition = "transform 0.3s ease";
		searchInput.value = ""
		populateOptions(defaultCities);
    
		if (dropdown.style.display === "block") {
			dropdown.style.display =  "none";
      dropdownIcon.style.transform = "rotate(180deg)";
		} else {
			dropdown.style.display = "block";
			dropdownIcon.style.transform = "rotate(0deg)";
		}
		
    searchInput.focus();
  });

  document.addEventListener("click", (e) => {
    if (!selectBox.contains(e.target) && !dropdown.contains(e.target)) {
      dropdown.style.display = "none";
			dropdownIcon.style.transform = "rotate(180deg)";
    }
  });

  searchInput.addEventListener("input", debounce(() => {
    searchCities(searchInput.value.trim());
  }, 300));

  togglePricingDisplay("São Paulo", false)
  await fetchAllCities();
  checkGeolocationPermission();

});

      function getUrlParams(urlOrQueryString) {
        if ((i = urlOrQueryString.indexOf("?")) >= 0) {
          const queryString = urlOrQueryString.substring(i + 1);
          if (queryString) {
            return _mapUrlParams(queryString);
          }
        }
    
        return {};
      }
    
      function _mapUrlParams(queryString) {
        return queryString.split("&").reduce(function (urlParams, urlParam) {
          urlParam = urlParam.split("=");
          if (
            Number.isInteger(parseInt(urlParam[1])) &&
            parseInt(urlParam[1]) == urlParam[1]
          ) {
            urlParams[urlParam[0]] = parseInt(urlParam[1]);
          } else {
            urlParams[urlParam[0]] = decodeURI(urlParam[1]);
          }
          return urlParams;
        }, {});
      }
    
      function updateQueryStringParameter(uri, key, value) {
        var re = new RegExp("([?&])" + key + "=.*?(&|$)", "i");
        var separator = uri.indexOf("?") !== -1 ? "&" : "?";
        if (uri.match(re)) {
          return uri.replace(re, "$1" + key + "=" + value + "$2");
        } else {
          return uri + separator + key + "=" + value;
        }
      }
